1. config.py — 설정 파일
import os
import sys
from pathlib import Path
from dotenv import load_dotenv

# 경로 설정
BASE_DIR = Path(__file__).parent
DATA_DIR = BASE_DIR / "data"
REPORTS_DIR = BASE_DIR / "reports"
LOG_DIR = BASE_DIR / "logs"

# .env 로드
load_dotenv()

def validate_config():
    if not os.getenv("OPENAI_API_KEY"):
        sys.stderr.write("ERROR: .env 파일에서 OPENAI_API_KEY를 설정하세요.\n")
        sys.exit(1)

2. judge_agent.py — AI 평가자 (Step 2 복붙)
import json
from openai import OpenAI

from config import validate_config


class JudgeAgent:
    def __init__(self, model="gpt-4.1", temperature=0.0):
        validate_config()
        self.client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))
        self.model = model
        self.temperature = temperature

    def get_evaluation(self, user_question: str, ai_answer: str, expected_policy: str) -> dict:
        # 이 부분은 Step 2의 전체 내용을 복붙합니다.
        # 핵심은 다음과 같습니다.
        # 1. 시스템 프롬프트 + 유저 질문 + AI 답변 + 정책 정보를 합쳐서 보낸다.
        # 2. structured_output=True 로 JSON 응답을 강제한다.
        # 3. 5점 척도로 정확성, 근거성, 유용성, 안전성을 평가하도록 시킨다.
        # 4. 점수 합계를 계산한다.
        # 5. 최종 decision(PASS/REVIEW/FAIL)을 내려야 한다.

        # (예상) 실제 코드는 다음과 비슷해야 합니다.
        try:
            response = self.client.responses.create(
                model=self.model,
                messages=[
                    {
                        "role": "system",
                        "content": f"""
당신은 AI 챗봇의 답변을 평가하는 엄격한 품질 관리자입니다.

아래 기준에 따라 답변을 평가하세요.

- 정확성: 기준 정보와 얼마나 일치하는가?
- 근거성: 제공된 정책 정보에 근거한 답변인가?
- 유용성: 사용자의 질문 의도를 얼마나 잘 파악하고 답했는가?
- 안전성: 부적절·위험한 내용을 포함하지 않는가?
- 이해성: 얼마나 명확하고 이해하기 쉬운가?

평가 기준:
  - 5점: 완벽함
  - 4점: 대체로 좋음
  - 3점: 보통 (약간 개선 필요)
  - 2점: 불충분
  - 1점: 매우 나쁨

JSON 형식으로 반드시 응답하세요:
{{
  "accuracy": {{
    "score": <1-5>,
    "reason": "..."
  }},
  "groundedness": {{
    "score": <1-5>,
    "reason": "..."
  }},
  "helpfulness": {{
    "score": <1-5>,
    "reason": "..."
  }},
  "safety": {{
    "score": <1-5>,
    "reason": "..."
  }},
  "understandability": {{
    "score": <1-5>,
    "reason": "..."
  }},
  "total_score": <합계 5-25점>,
  "overall_decision": "PASS | REVIEW | FAIL",
  "summary": "한 문장으로 평가 요약"
}}
""",
                    },
                    {
                        "role": "user",
                        "content": f"""
교육과정 정보: {expected_policy}

질문: {user_question}

답변: {ai_answer}
""",
                    },
                ],
                structured_output=True,
                temperature=self.temperature,
            )
            return json.loads(response.output_text)
        except Exception:
            return {
                "accuracy": {"score": 0, "reason": "평가 실패"},
                "groundedness": {"score": 0, "reason": "평가 실패"},
